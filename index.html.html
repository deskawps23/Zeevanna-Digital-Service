<script>

    // ===== DATABASE SEDERHANA (Simpan di HP) =====

    let data = JSON.parse(localStorage.getItem('zeevanna_data')) || {

        orders: [],

        customers: [],

        freelancers: [],

        currentUser: null,

        orderCounter: 1

    };

    function simpanData() {

        localStorage.setItem('zeevanna_data', JSON.stringify(data));

    }

    // ===== LOGIN =====

    window.login = function() {

        const nama = document.getElementById('inputNama').value;

        const role = document.getElementById('inputRole').value.toLowerCase();

        if (!nama) return alert('Isi nama dulu!');

        data.currentUser = { nama, role };

        simpanData();

        if (role === 'freelancer') {

            document.getElementById('page-login').classList.add('hide');

            document.getElementById('page-freelancer').classList.remove('hide');

            document.getElementById('page-customer').classList.add('hide');

            document.getElementById('namaFreelancer').innerText = nama;

            loadFreelancer();

        } else {

            document.getElementById('page-login').classList.add('hide');

            document.getElementById('page-customer').classList.remove('hide');

            document.getElementById('page-freelancer').classList.add('hide');

            document.getElementById('totalPesanCustomer').innerText = data.orders.filter(o => o.customer === nama).length;

            loadFreelancer(); // refresh pending orders

        }

    };

    // ===== LOGOUT =====

    window.logout = function() {

        data.currentUser = null;

        simpanData();

        document.getElementById('page-login').classList.remove('hide');

        document.getElementById('page-customer').classList.add('hide');

        document.getElementById('page-freelancer').classList.add('hide');

    };

    // ===== TAB NAVIGASI =====

    window.tab = function(id) {

        document.querySelectorAll('.menu button').forEach(b => b.classList.remove('aktif'));

        event.target.classList.add('aktif');

        document.querySelectorAll('#page-customer > div:not(.menu)').forEach(d => d.classList.add('hide'));

        document.querySelectorAll('#page-freelancer > div:not(.menu)').forEach(d => d.classList.add('hide'));

        if (id === 'pesan') document.getElementById('tab-pesan').classList.remove('hide');

        if (id === 'track') document.getElementById('tab-track').classList.remove('hide');

        if (id === 'dashboard') document.getElementById('tab-dashboard').classList.remove('hide');

    };

    // ===== CUSTOMER: BUAT PESANAN =====

    window.buatPesanan = function() {

        const user = data.currentUser;

        if (!user || user.role === 'freelancer') return alert('Login sebagai customer dulu!');

        const harga = parseInt(document.getElementById('pilihJenis').value);

        const jumlah = parseInt(document.getElementById('inputJumlah').value) || 1;

        const alamat = document.getElementById('inputAlamat').value;

        const total = harga * jumlah;

        const id = data.orderCounter++;

        const order = {

            id: id,

            customer: user.nama,

            item: document.getElementById('pilihJenis').options[document.getElementById('pilihJenis').selectedIndex].text,

            qty: jumlah,

            total: total,

            alamat: alamat,

            status: 'pending', // pending, diproses, selesai

            freelancer: null

        };

        data.orders.push(order);

        simpanData();

        document.getElementById('hasilPesan').style.display = 'block';

        document.getElementById('hasilPesan').innerHTML = `✅ Pesanan #${id} berhasil! Total: Rp ${total.toLocaleString()}`;

        setTimeout(() => document.getElementById('hasilPesan').style.display = 'none', 3000);

        document.getElementById('totalPesanCustomer').innerText = data.orders.filter(o => o.customer === user.nama).length;

        loadFreelancer(); // Update dashboard freelancer

    };

    // ===== CUSTOMER: TRACKING =====

    window.tracking = function() {

        const id = parseInt(document.getElementById('inputIdPesan').value);

        const order = data.orders.find(o => o.id === id);

        const div = document.getElementById('hasilTrack');

        if (!order) {

            div.innerHTML = '<div class="card" style="background:#ffebee;"><p>❌ Pesanan tidak ditemukan</p></div>';

            return;

        }

        let statusLabel = '⏳ Menunggu Mitra';

        if (order.status === 'diproses') statusLabel = '🧺 Sedang Disetrika';

        if (order.status === 'selesai') statusLabel = '✅ Selesai Diantar';

        div.innerHTML = `

            <div class="card">

                <h4>#${order.id} - ${order.item}</h4>

                <p>Jumlah: ${order.qty} | Total: Rp ${order.total.toLocaleString()}</p>

                <p>Alamat: ${order.alamat}</p>

                <p><strong>Status:</strong> ${statusLabel}</p>

                <p>Mitra: ${order.freelancer || 'Belum ada'}</p>

            </div>

        `;

    };

    // ===== FREELANCER: LOAD DASHBOARD =====

    function loadFreelancer() {

        const pendingDiv = document.getElementById('daftarPending');

        const tugasDiv = document.getElementById('daftarTugas');

        const user = data.currentUser;

        let saldo = 0;

        // Hitung saldo dari pesanan selesai

        data.orders.forEach(o => { if (o.freelancer === user?.nama && o.status === 'selesai') saldo += o.total; });

        document.getElementById('saldoFreelancer').innerText = saldo.toLocaleString();

        // Pesanan Pending

        const pending = data.orders.filter(o => o.status === 'pending');

        if (pending.length === 0) {

            pendingDiv.innerHTML = '<div class="card"><p class="status">📭 Tidak ada pesanan pending.</p></div>';

        } else {

            let html = '<h4>📩 Pesanan Tersedia</h4>';

            pending.forEach(o => {

                html += `

                    <div class="card">

                        <p><strong>#${o.id}</strong> - ${o.customer} (${o.item} x${o.qty})</p>

                        <p>Alamat: ${o.alamat}</p>

                        <p>Total: Rp ${o.total.toLocaleString()}</p>

                        <button class="btn" style="padding:8px; font-size:14px;" onclick="ambilPesanan(${o.id})">✅ Ambil Pesanan</button>

                    </div>

                `;

            });

            pendingDiv.innerHTML = html;

        }

        // Tugas Saya

        const tugas = data.orders.filter(o => o.freelancer === user?.nama && o.status !== 'selesai');

        if (tugas.length === 0) {

            tugasDiv.innerHTML = '<div class="card"><p class="status">🛠️ Tidak ada tugas aktif.</p></div>';

        } else {

            let html = '<h4>🛠️ Tugas Saya</h4>';

            tugas.forEach(o => {

                let btn = '';

                if (o.status === 'pending' || o.status === 'diproses') {

                    btn = `<button class="btn" style="padding:8px; font-size:14px;" onclick="updateStatus(${o.id}, 'diproses')">🧺 Mulai Setrika</button>`;

                }

                if (o.status === 'diproses') {

                    btn = `<button class="btn" style="padding:8px; font-size:14px;" onclick="updateStatus(${o.id}, 'selesai')">🚚 Antar Selesai</button>`;

                }

                html += `

                    <div class="card">

                        <p><strong>#${o.id}</strong> - ${o.customer} (${o.item} x${o.qty})</p>

                        <p>Status: ${o.status}</p>

                        ${btn}

                    </div>

                `;

            });

            tugasDiv.innerHTML = html;

        }

    }

    // ===== FREELANCER: AMBIL PESANAN =====

    window.ambilPesanan = function(id) {

        const order = data.orders.find(o => o.id === id);

        if (!order) return alert('Pesanan tidak ditemukan');

        if (!data.currentUser) return alert('Login dulu');

        order.freelancer = data.currentUser.nama;

        order.status = 'diproses';

        simpanData();

        loadFreelancer();

        alert('✅ Berhasil ambil pesanan!');

    };

    // ===== FREELANCER: UPDATE STATUS =====

    window.updateStatus = function(id, status) {

        const order = data.orders.find(o => o.id === id);

        if (!order) return alert('Pesanan tidak ditemukan');

        order.status = status;

        simpanData();

        loadFreelancer();

        alert('✅ Status berhasil diupdate!');

    };

    // ===== CEK SESSION SAAT LOAD =====

    (function init() {

        if (data.currentUser) {

            if (data.currentUser.role === 'freelancer') {

                document.getElementById('page-login').classList.add('hide');

                document.getElementById('page-freelancer').classList.remove('hide');

                document.getElementById('page-customer').classList.add('hide');

                document.getElementById('namaFreelancer').innerText = data.currentUser.nama;

                loadFreelancer();

            } else {

                document.getElementById('page-login').classList.add('hide');

                document.getElementById('page-customer').classList.remove('hide');

                document.getElementById('page-freelancer').classList.add('hide');

                document.getElementById('totalPesanCustomer').innerText = data.orders.filter(o => o.customer === data.currentUser.nama).length;

                loadFreelancer();

            }

        }

    })();

</script>
